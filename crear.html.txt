<script>
document.addEventListener("DOMContentLoaded", function() {
    console.log("🚀 Página cargada");
    
    let formCount = {{ formset.total_forms }};
    const addFormBtn = document.getElementById("add-form");
    const itemsContainer = document.getElementById("itemsContainer");
    const mensajeInicial = document.getElementById("mensaje-inicial");
    const totalGeneral = document.getElementById("total-general");
    
    console.log("📋 FormCount inicial:", formCount);
    
    const primeraFila = document.querySelector(".formset-row");
    if (primeraFila) {
        primeraFila.style.display = 'none';
        console.log("👁️ Fila inicial oculta");
    }
    
    addFormBtn.addEventListener("click", function() {
        console.log("➕ Clic en Agregar Item");
        
        const filasVisibles = document.querySelectorAll(".formset-row:not([style*='display: none'])").length;
        console.log("Filas visibles:", filasVisibles);
        
        if (filasVisibles >= 10) {
            alert("Límite de 10 productos alcanzado");
            return;
        }
        
        if (!primeraFila) {
            console.error("❌ No hay fila base");
            return;
        }
        
        const nuevaFila = primeraFila.cloneNode(true);
        console.log("📋 Fila clonada");
        
        const newIndex = formCount;
        nuevaFila.setAttribute('data-form-index', newIndex);
        
        nuevaFila.querySelectorAll("input, select").forEach(function(element) {
            const name = element.getAttribute('name');
            const id = element.getAttribute('id');
            
            if (name) {
                const newName = name.replace(/detalles-\d+-/, "detalles-" + newIndex + "-");
                element.setAttribute('name', newName);
            }
            
            if (id) {
                const newId = id.replace(/detalles-\d+|id_detalles-\d+/, function(match) {
                    return match.startsWith('id_') ? "id_detalles-" + newIndex : "detalles-" + newIndex;
                });
                element.setAttribute('id', newId);
            }
        });
        
        const productoNombre = nuevaFila.querySelector("div[id^='producto-nombre-']");
        if (productoNombre) {
            productoNombre.setAttribute('id', "producto-nombre-" + newIndex);
        }
        
        const subtotalInput = nuevaFila.querySelector("input[id^='subtotal-']");
        if (subtotalInput) {
            subtotalInput.setAttribute('id', "subtotal-" + newIndex);
        }
        
        nuevaFila.querySelectorAll("input[type='number']").forEach(el => el.value = '');
        nuevaFila.querySelectorAll("select").forEach(el => el.selectedIndex = 0);
        nuevaFila.querySelectorAll("input[type='text']").forEach(el => {
            if (!el.readOnly) el.value = '';
        });
        nuevaFila.querySelectorAll("input[type='checkbox']").forEach(el => el.checked = false);
        nuevaFila.querySelectorAll("div[id^='producto-nombre-']").forEach(el => {
            el.textContent = '';
            el.style.display = 'none';
        });
        
        nuevaFila.style.display = '';
        document.getElementById("formset-container").appendChild(nuevaFila);
        
        formCount++;
        document.querySelector("input[name='detalles-TOTAL_FORMS']").value = formCount;
        
        const numeroInput = nuevaFila.querySelector("input[readonly]");
        if (numeroInput) {
            numeroInput.value = filasVisibles + 1;
        }
        
        itemsContainer.style.display = 'block';
        mensajeInicial.style.display = 'none';
        totalGeneral.style.display = '';
        
        console.log("✅ Fila agregada, formCount:", formCount);
    });
    
    function calcularSubtotales() {
        let total = 0;
        let hayProductos = false;
        
        document.querySelectorAll(".formset-row:not([style*='display: none'])").forEach(function(row) {
            const cantidad = row.querySelector("input[name*='cantidad']");
            const precio = row.querySelector("input[name*='precio_unitario']");
            const subtotal = row.querySelector("input[name*='subtotal']");
            
            if (cantidad && precio && subtotal) {
                const cant = parseFloat(cantidad.value) || 0;
                const prec = parseFloat(precio.value) || 0;
                const sub = cant * prec;
                
                subtotal.value = sub.toFixed(2);
                total += sub;
                
                if (cant > 0 && prec > 0) hayProductos = true;
            }
        });
        
        const totalInput = document.getElementById("total-general-input");
        if (totalInput) {
            totalInput.value = "Bs. " + total.toFixed(2);
        }
        
        console.log("💰 Total calculado:", total);
    }
    
    document.addEventListener('input change', function(e) {
        if (e.target.matches("input[name*='cantidad'], input[name*='precio_unitario']")) {
            calcularSubtotales();
        }
    });
    
    document.getElementById("crear-producto-btn").addEventListener("click", function() {
        console.log("📦 Crear producto");
        const formData = new FormData(document.getElementById('compra-form'));
        const dataStr = new URLSearchParams(formData).toString();
        sessionStorage.setItem('compraFormData', dataStr);
        window.location.href = "{% url 'inventario:crear' %}?return_to=compras&compra_form_data=" + encodeURIComponent(dataStr);
    });
    
    calcularSubtotales();
    console.log("✅ Sistema inicializado");
});
</script>
